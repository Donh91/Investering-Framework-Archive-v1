RUN python - <<'PY'
import hashlib
import io
from pathlib import Path
import tarfile
import urllib.request

version = '24.11.1'
expected = '60e3b0a8500819514aca603487c254298cd776de0698d3cd08f11dba5b8289a8'
url = f'https://nodejs.org/dist/v{version}/node-v{version}-linux-x64.tar.xz'
with urllib.request.urlopen(url, timeout=120) as source:
    data = source.read(50_000_000)
if hashlib.sha256(data).hexdigest() != expected:
    raise SystemExit('NODE_TARBALL_DIGEST_UNVERIFIED')
root = Path('/opt/node')
root.mkdir(parents=True, exist_ok=True)
with tarfile.open(fileobj=io.BytesIO(data), mode='r:xz') as archive:
    archive.extractall(root, filter='data')
actual = root / f'node-v{version}-linux-x64'
if not actual.is_dir():
    raise SystemExit('NODE_BIN_NOT_FOUND')
actual.rename(root / 'current')
PY
RUN node --version && python -m pytest --version
